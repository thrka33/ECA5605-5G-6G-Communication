%% INDUSTRIAL 5G NETWORK DESIGN AND VALIDATION
% Smart Factory Private 5G Performance Evaluation
% Design A: Conventional 5G
% Design B: Proposed Massive-MIMO + Beamforming 5G

clear;
clc;
close all;

%% ================================================================
% 1. SYSTEM PARAMETERS
% ================================================================

fc = 3.5e9;                 % Carrier frequency (Hz)
c = 3e8;                    % Speed of light (m/s)
lambda = c/fc;

bandwidth = 100e6;          % 100 MHz
noiseFigure = 7;            % dB
txPower_dBm = 30;           % 1 W transmit power
thermalNoise_dBm = -174;    % dBm/Hz

% Industrial factory dimensions
factoryLength = 200;        % meters
factoryWidth  = 100;        % meters

% Distances for simulation
distance = 10:10:200;

% Device densities
deviceCounts = [50 100 200 300 400 500];

% Traffic load
trafficLoad = [20 40 60 80 100];   % percentage

%% ================================================================
% 2. NOISE POWER
% ================================================================

noisePower_dBm = thermalNoise_dBm + ...
                 10*log10(bandwidth) + noiseFigure;

noisePower_W = 10.^((noisePower_dBm-30)/10);

%% ================================================================
% 3. DESIGN PARAMETERS
% ================================================================

% Design A - Conventional 5G
Nt_A = 2;                   % Transmit antennas
Nr_A = 2;                   % Receive antennas
beamformingGain_A = 0;      % dB
resourceEfficiency_A = 0.75;
energyPerBit_A = 1.5e-8;    % J/bit

% Design B - Proposed Industrial 5G
Nt_B = 64;                  % Massive MIMO antennas
Nr_B = 8;
beamformingGain_B = 10*log10(Nt_B);
resourceEfficiency_B = 0.92;
energyPerBit_B = 8e-9;

%% ================================================================
% 4. PATH LOSS MODEL
% ================================================================

pathLoss = zeros(size(distance));

for k = 1:length(distance)

    d = distance(k);

    % Free-space path loss
    pathLoss(k) = 20*log10(4*pi*d/lambda);

end

%% ================================================================
% 5. SINR CALCULATION
% ================================================================

SINR_A_dB = zeros(size(distance));
SINR_B_dB = zeros(size(distance));

for k = 1:length(distance)

    receivedPower_A_dBm = ...
        txPower_dBm - pathLoss(k) + beamformingGain_A;

    receivedPower_B_dBm = ...
        txPower_dBm - pathLoss(k) + beamformingGain_B;

    interference_A_dB = 8;
    interference_B_dB = 4;

    SINR_A_dB(k) = receivedPower_A_dBm - ...
        10*log10(10^(noisePower_dBm/10) + ...
        10^(interference_A_dB/10));

    SINR_B_dB(k) = receivedPower_B_dBm - ...
        10*log10(10^(noisePower_dBm/10) + ...
        10^(interference_B_dB/10));

end

%% ================================================================
% 6. BER CALCULATION
% ================================================================

% Convert SINR to linear
SINR_A = 10.^(SINR_A_dB/10);
SINR_B = 10.^(SINR_B_dB/10);

% BPSK theoretical BER approximation
BER_A = 0.5 * erfc(sqrt(SINR_A));
BER_B = 0.5 * erfc(sqrt(SINR_B));

%% ================================================================
% 7. SPECTRAL EFFICIENCY
% ================================================================

spectralEfficiency_A = log2(1 + SINR_A) * resourceEfficiency_A;
spectralEfficiency_B = log2(1 + SINR_B) * resourceEfficiency_B;

% Limit practical spectral efficiency
spectralEfficiency_A = min(spectralEfficiency_A, 8);
spectralEfficiency_B = min(spectralEfficiency_B, 12);

%% ================================================================
% 8. THROUGHPUT
% ================================================================

throughput_A = bandwidth .* spectralEfficiency_A / 1e6;
throughput_B = bandwidth .* spectralEfficiency_B / 1e6;

% Mbps

%% ================================================================
% 9. LATENCY MODEL
% ================================================================

processingLatency = 1;      % ms
transmissionLatency = 1;    % ms
queueLatency_A = 4;         % ms
queueLatency_B = 2;         % ms

latency_A = processingLatency + ...
            transmissionLatency + queueLatency_A + ...
            distance/(c/1000);

latency_B = processingLatency + ...
            transmissionLatency + queueLatency_B + ...
            distance/(c/1000);

%% ================================================================
% 10. RELIABILITY
% ================================================================

reliability_A = 1 - BER_A;
reliability_B = 1 - BER_B;

reliability_A = min(reliability_A,0.999999);
reliability_B = min(reliability_B,0.9999999);

%% ================================================================
% 11. ENERGY EFFICIENCY
% ================================================================

energyEfficiency_A = throughput_A*1e6 ./ energyPerBit_A;
energyEfficiency_B = throughput_B*1e6 ./ energyPerBit_B;

%% ================================================================
% 12. COVERAGE ANALYSIS
% ================================================================

coverageThreshold = 5;      % minimum SINR in dB

coverage_A = distance(SINR_A_dB >= coverageThreshold);
coverage_B = distance(SINR_B_dB >= coverageThreshold);

if isempty(coverage_A)
    maxCoverage_A = 0;
else
    maxCoverage_A = max(coverage_A);
end

if isempty(coverage_B)
    maxCoverage_B = 0;
else
    maxCoverage_B = max(coverage_B);
end

%% ================================================================
% 13. DEVICE DENSITY ANALYSIS
% ================================================================

densityThroughput_A = zeros(size(deviceCounts));
densityThroughput_B = zeros(size(deviceCounts));

densityLatency_A = zeros(size(deviceCounts));
densityLatency_B = zeros(size(deviceCounts));

densityReliability_A = zeros(size(deviceCounts));
densityReliability_B = zeros(size(deviceCounts));

for k = 1:length(deviceCounts)

    N = deviceCounts(k);

    % Increased contention with device density
    contention_A = 1 + 0.004*N;
    contention_B = 1 + 0.0015*N;

    densityThroughput_A(k) = ...
        max(throughput_A(1)/contention_A,1);

    densityThroughput_B(k) = ...
        max(throughput_B(1)/contention_B,1);

    densityLatency_A(k) = ...
        5 + 0.02*N;

    densityLatency_B(k) = ...
        3 + 0.008*N;

    densityReliability_A(k) = ...
        exp(-0.002*N);

    densityReliability_B(k) = ...
        exp(-0.0008*N);

end

%% ================================================================
% 14. TRAFFIC LOAD ANALYSIS
% ================================================================

trafficThroughput_A = zeros(size(trafficLoad));
trafficThroughput_B = zeros(size(trafficLoad));

trafficLatency_A = zeros(size(trafficLoad));
trafficLatency_B = zeros(size(trafficLoad));

for k = 1:length(trafficLoad)

    load = trafficLoad(k);

    trafficThroughput_A(k) = ...
        throughput_A(1) * (load/100) * ...
        (1 - 0.002*load);

    trafficThroughput_B(k) = ...
        throughput_B(1) * (load/100) * ...
        (1 - 0.001*load);

    trafficLatency_A(k) = ...
        5 + 0.06*load;

    trafficLatency_B(k) = ...
        3 + 0.025*load;

end
